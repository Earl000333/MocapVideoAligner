"""
Dynamics-based mocap <-> plantar-pressure temporal alignment (delta_t2).

Design notes
------------
The legacy curve (-min_foot_height, min-max normalised) saturates during
stance, so the cross-correlation peak is broad and the lag is poorly
localised.  Here both sides are converted into an estimate of the SAME
physical quantity, the vertical ground reaction force in body weights
(BW), then compared:

    mocap  side : vGRF/BW = a_com_z / g + 1        (whole-body CoM, 2nd derivative)
    tactile side: vGRF/BW = p_total / mean(p_total)

Body mass cancels on both sides, so no subject mass is required.

Two rules that dominate the achievable accuracy:
  * every filter must be zero-phase (filtfilt).  A causal filter injects a
    group delay that lands directly in delta_t2 as a systematic bias.
  * both signals must pass through the IDENTICAL filter before comparison.
    The insole sees the heel-strike impact transient, the doubly
    differentiated CoM does not; matching the bandwidth removes that bias.

Reference implementation only.  Wire into the new tab as a separate module;
do not touch the existing visual-mocap path.
"""

from __future__ import annotations

import numpy as np
from scipy.interpolate import CubicSpline
from scipy.signal import butter, filtfilt

G = 9.80665

# ---------------------------------------------------------------------------
# 1. Whole-body centre of mass
# ---------------------------------------------------------------------------

# Winter (2009) segment mass fractions and CoM position as a fraction of the
# segment length measured from the proximal joint.  Fractions sum to 1.0.
# (proximal, distal, mass_fraction, com_ratio)
SEGMENT_TABLE = [
    ("Spine",       "Neck",        0.497, 0.500),   # trunk
    ("Neck",        "Head",        0.081, 0.500),   # head + neck
    ("LeftArm",     "LeftForeArm", 0.028, 0.436),
    ("RightArm",    "RightForeArm", 0.028, 0.436),
    ("LeftForeArm", "LeftHand",    0.016, 0.430),
    ("RightForeArm", "RightHand",  0.016, 0.430),
    ("LeftHand",    "LeftHandEnd", 0.006, 0.506),
    ("RightHand",   "RightHandEnd", 0.006, 0.506),
    ("LeftUpLeg",   "LeftLeg",     0.100, 0.433),
    ("RightUpLeg",  "RightLeg",    0.100, 0.433),
    ("LeftLeg",     "LeftFoot",    0.0465, 0.433),
    ("RightLeg",    "RightFoot",   0.0465, 0.433),
    ("LeftFoot",    "LeftToeBase", 0.0145, 0.500),
    ("RightFoot",   "RightToeBase", 0.0145, 0.500),
]


def body_com(joint_xyz: dict[str, np.ndarray],
             table=SEGMENT_TABLE) -> np.ndarray:
    """Whole-body CoM trajectory, shape (T, 3).

    joint_xyz maps a joint name to an (T, 3) array of world positions.
    Segments whose joints are absent are dropped and the remaining mass
    fractions are renormalised, so a partial skeleton still works.
    """
    total, acc = 0.0, None
    for prox, dist, mass, ratio in table:
        if prox not in joint_xyz or dist not in joint_xyz:
            continue
        p, d = joint_xyz[prox], joint_xyz[dist]
        seg_com = p + ratio * (d - p)
        acc = seg_com * mass if acc is None else acc + seg_com * mass
        total += mass
    if acc is None or total == 0.0:
        raise ValueError("no usable segment found in joint_xyz")
    return acc / total


# ---------------------------------------------------------------------------
# 2. Filtering and differentiation
# ---------------------------------------------------------------------------

def zero_phase_lowpass(x: np.ndarray, fs: float, fc: float,
                       order: int = 4) -> np.ndarray:
    """Zero-phase Butterworth low pass, applied along axis 0."""
    if fc >= fs / 2.0:
        return np.asarray(x, dtype=float)
    b, a = butter(order, fc / (fs / 2.0), btype="low")
    return filtfilt(b, a, np.asarray(x, dtype=float), axis=0)


def second_derivative(x: np.ndarray, fs: float) -> np.ndarray:
    """Central second difference; filter BEFORE calling this."""
    d = np.gradient(np.gradient(np.asarray(x, dtype=float), axis=0), axis=0)
    return d * (fs ** 2)


def predict_vgrf_bw(com_xyz: np.ndarray, fs: float, vertical: int = 2,
                    fc: float = 8.0) -> np.ndarray:
    """Predicted total vertical GRF in body weights from the CoM trajectory."""
    com_f = zero_phase_lowpass(com_xyz, fs, fc)
    a_v = second_derivative(com_f[:, vertical], fs)
    return a_v / G + 1.0


def measured_vgrf_bw(p_total: np.ndarray, fs: float,
                     fc: float = 8.0) -> np.ndarray:
    """Insole total, band-matched to the mocap side and scaled to body weights.

    Over an integer number of gait cycles the mean vertical GRF equals body
    weight, which supplies the scale without any calibration trial.
    """
    p = zero_phase_lowpass(np.asarray(p_total, dtype=float), fs, fc)
    m = float(np.mean(p))
    return p / m if m > 0 else p


# ---------------------------------------------------------------------------
# 3. Foot contact and the double-support split (cross-check channel only)
# ---------------------------------------------------------------------------

def foot_contact(heel_xyz: np.ndarray, toe_xyz: np.ndarray, fs: float,
                 vertical: int = 2, speed_thr: float = 0.35,
                 height_band: float = 0.03) -> np.ndarray:
    """Boolean stance mask from foot kinematics (velocity + height gate).

    speed_thr is in metres per second, height_band in metres above the
    session-wide lowest point of that foot.  Units must match the BVH.
    """
    pos = np.minimum(heel_xyz[:, vertical], toe_xyz[:, vertical])
    pos = zero_phase_lowpass(pos, fs, min(10.0, fs / 2.5))
    vel = np.abs(np.gradient(pos) * fs)
    floor = np.percentile(pos, 2.0)
    return (pos < floor + height_band) & (vel < speed_thr)


def split_vgrf(vgrf: np.ndarray, contact_l: np.ndarray,
               contact_r: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """Distribute the total onto both feet.

    Single support gets the whole load.  Double support uses a smoothstep
    transfer from the trailing to the leading foot, which is the cheap
    stand-in for a real CoP model.  Flight gets zero.
    """
    n = len(vgrf)
    wl = np.zeros(n)
    wl[contact_l & ~contact_r] = 1.0
    wl[~contact_l & contact_r] = 0.0
    both = contact_l & contact_r
    idx = 0
    while idx < n:
        if not both[idx]:
            idx += 1
            continue
        j = idx
        while j < n and both[j]:
            j += 1
        # leading foot is the one that was airborne just before the interval
        lead_left = bool(idx > 0 and not contact_l[idx - 1])
        s = np.linspace(0.0, 1.0, j - idx)
        ramp = 3 * s ** 2 - 2 * s ** 3
        wl[idx:j] = ramp if lead_left else 1.0 - ramp
        idx = j
    return vgrf * wl, vgrf * (1.0 - wl)


# ---------------------------------------------------------------------------
# 4. Lag estimation
# ---------------------------------------------------------------------------

def _resample(t: np.ndarray, y: np.ndarray, fs_out: float):
    grid = np.arange(t[0], t[-1], 1.0 / fs_out)
    return grid, CubicSpline(t, y)(grid)


def xcorr_lag(a: np.ndarray, b: np.ndarray, fs: float, t_center: float,
              half_window_s: float, fs_work: float = 500.0,
              use_derivative: bool = True) -> tuple[float, float]:
    """Normalised cross-correlation lag of b relative to a, in seconds.

    Both signals are band limited well below fs_work, so spline upsampling
    is close to exact and sub-sample interpolation of the peak is legitimate.
    Correlating the time derivative sharpens the peak, at the cost of some
    noise gain.
    """
    t = np.arange(len(a)) / fs
    _, ai = _resample(t, a, fs_work)
    _, bi = _resample(np.arange(len(b)) / fs, b, fs_work)
    if use_derivative:
        ai, bi = np.gradient(ai), np.gradient(bi)
    ai = (ai - ai.mean()) / (ai.std() + 1e-12)
    bi = (bi - bi.mean()) / (bi.std() + 1e-12)

    max_lag = int(round((abs(t_center) + half_window_s) * fs_work)) + 2
    full = np.correlate(ai, bi, mode="full") / min(len(ai), len(bi))
    lags = (np.arange(full.size) - (len(bi) - 1)) / fs_work
    keep = np.abs(lags - t_center) <= half_window_s
    if not keep.any():
        keep = np.abs(lags) <= max_lag / fs_work
    seg, seg_lags = full[keep], lags[keep]

    k = int(np.argmax(seg))
    peak = float(seg[k])
    lag = float(seg_lags[k])
    if 0 < k < len(seg) - 1:                       # parabolic sub-sample peak
        y0, y1, y2 = seg[k - 1], seg[k], seg[k + 1]
        denom = y0 - 2 * y1 + y2
        if denom != 0:
            lag += 0.5 * (y0 - y2) / denom / fs_work
    return lag, peak


def onset_times(sig: np.ndarray, fs: float, rel_thr: float = 0.15,
                min_gap_s: float = 0.35) -> np.ndarray:
    """Rising threshold crossings at a fixed fraction of the stance peak.

    The same functional is applied to both signals, so the definitional part
    of the bias cancels between them.
    """
    s = np.asarray(sig, dtype=float)
    lo, hi = np.percentile(s, 5), np.percentile(s, 95)
    thr = lo + rel_thr * (hi - lo)
    above = s > thr
    cross = np.where(~above[:-1] & above[1:])[0]
    out, last = [], -np.inf
    for i in cross:
        y0, y1 = s[i], s[i + 1]
        t = (i + (thr - y0) / (y1 - y0 + 1e-12)) / fs
        if t - last >= min_gap_s:
            out.append(t)
            last = t
    return np.asarray(out)


def refine_by_events(a: np.ndarray, b: np.ndarray, fs: float,
                     coarse_lag: float, tol_s: float = 0.12):
    """Median offset over matched events, plus scatter and clock drift.

    Returns (delta, mad, drift_ppm, n_matched).  drift_ppm is the slope of
    the per-event offset against time; a non-zero slope means the two clocks
    run at different rates and a single scalar delta cannot hold over a long
    session.
    """
    ea, eb = onset_times(a, fs), onset_times(b, fs)
    if ea.size == 0 or eb.size == 0:
        return coarse_lag, np.nan, np.nan, 0
    pairs = []
    for t in ea:
        k = int(np.argmin(np.abs(eb - (t + coarse_lag))))
        d = eb[k] - t
        if abs(d - coarse_lag) <= tol_s:
            pairs.append((t, d))
    if len(pairs) < 3:
        return coarse_lag, np.nan, np.nan, len(pairs)
    tt = np.array([p[0] for p in pairs])
    dd = np.array([p[1] for p in pairs])
    delta = float(np.median(dd))
    mad = float(np.median(np.abs(dd - delta)))
    slope = float(np.polyfit(tt, dd, 1)[0]) if tt.ptp() > 5.0 else 0.0
    return delta, mad, slope * 1e6, len(pairs)


# ---------------------------------------------------------------------------
# 5. Top level
# ---------------------------------------------------------------------------

def estimate_delta_t2(joint_xyz: dict[str, np.ndarray], fs_mocap: float,
                      left_sum: np.ndarray, right_sum: np.ndarray,
                      fs_pressure: float, t_coarse: float = 0.0,
                      half_window_s: float = 0.20, vertical: int = 2,
                      fc: float = 8.0) -> dict:
    """Both streams are assumed already resampled to a common reference fps."""
    if abs(fs_mocap - fs_pressure) > 1e-6:
        raise ValueError("resample both streams to reference_fps first")
    fs = fs_mocap

    com = body_com(joint_xyz)
    pred_total = predict_vgrf_bw(com, fs, vertical=vertical, fc=fc)
    meas_total = measured_vgrf_bw(np.asarray(left_sum) + np.asarray(right_sum),
                                  fs, fc=fc)
    n = min(len(pred_total), len(meas_total))
    pred_total, meas_total = pred_total[:n], meas_total[:n]

    lag, peak = xcorr_lag(pred_total, meas_total, fs, t_coarse, half_window_s)
    delta, mad, drift_ppm, n_ev = refine_by_events(pred_total, meas_total,
                                                   fs, lag)

    shifted = CubicSpline(np.arange(n) / fs, meas_total,
                          extrapolate=False)(np.arange(n) / fs - delta)
    ok = np.isfinite(shifted)
    ss_res = float(np.sum((pred_total[ok] - shifted[ok]) ** 2))
    ss_tot = float(np.sum((pred_total[ok] - pred_total[ok].mean()) ** 2))
    r2 = 1.0 - ss_res / ss_tot if ss_tot > 0 else np.nan
    se = mad * 1.4826 / np.sqrt(n_ev) if n_ev > 0 else np.nan

    return dict(delta_t2=delta, delta_t2_xcorr=lag, xcorr_peak=peak,
                event_mad_s=mad, event_count=n_ev, standard_error_s=se,
                clock_drift_ppm=drift_ppm, r2_total=r2, t_coarse=t_coarse,
                search_window_ms=half_window_s * 1000.0, filter_cutoff_hz=fc)
